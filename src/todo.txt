  for (uint8_t i = 0; i < NUM_ENC; i++) {

//bts.cpp
#include "bts.h"
#include "../config/config.h"

static void setMotor(uint8_t pinFwd, uint8_t pinRev, int v, bool invertido){ //limite y sentido
    if(invertido){
        v = -v;
    }
    v = constrain(v, -255, 255);

    if(v > 0){
        ledcWrite(pinWrite(pinRev, 0));
        ledcWrite(pinWrite(pinFwd, v));
    }
    else if (v < 0){
        ledcWrite(pinWrite(pinRev, -v));
        ledcWrite(pinWrite(pinFwd, 0));
    }
    else{
        ledcWrite(pinWrite(pinRev, 0));
        ledcWrite(pinWrite(pinFwd, 0));
    }
}
void iniciarMotores(){

    const uint8_t pwmPines[4] = = {MOT_IZQ_RPWM, MOT_IZQ_LPWM, MOT_DER_RPWM, MOT_DER_LPWM};
    const uint8_t enPines[4]  = {MOT_IZQ_R_EN, MOT_IZQ_L_EN, MOT_DER_R_EN, MOT_DER_L_EN};

    for(uint8_t i = 0; i < 4; i++){// LOW lo mas antes posible para que no salgan volando solos
        pinMode(pwmPines[i], OUTPUT);
        digitalWrite(pwmPines[i], LOW)
    }

    for(uint8_t i = 0; i < 4; i++){ //enables en high, sin enable no mueve
        pinMode(enPines[i], OUTPUT);
        digitalWrite(enPines[i], HIGH);
    }

    for (uint8_t i = 0; i < 4; i++) {//activa el PWM (core 3.x)
    ledcAttach(pwmPines[i], MOT_PWM_FREQ, MOT_PWM_BITS);
    }

    detenerMotores();
}

void motores(int izq, int der){
    setMotor(MOT_IZQ_RPWM, MOT_IZQ_LPWM, izq, MOT_IZQ_INVERTIDO);
    setMotor(MOT_DER_RPWM, MOT_DER_LPWM, der, MOT_DER_INVERTIDO);
}

void detenerMotores() {
  motores(0, 0);
}

// encoders.cpp
#include "encoders.h"
#include "../config/config.h"
#include <Wire.h>


static uint16_t angulo[NUM_ENC];
static bool activo[NUM_ENC];
static TwoWire* bus[NUM_ENC]; //direccion de un bus I2C

static bool leerRaw(TwoWire*& w, uint16_t& out){// manera simple de no usar #include <AS5600.h> en el codigo 
    w.beginTransmission(AS5600_DIR);
    w.write(0x0C);// registro del angulo, envia dato
    if (w.endTransmission(false) != 0) return false;// no contesta
    if (w.requestFrom((uint8_t)AS5600_DIR, (uint8_t)2) != 2) return false;

    uint16_t alto = w.read();
    uint16_t bajo = w.read();
    out = ((alto << 8) | bajo) & 0x0FFF;// solo 12 bits
    return true;
}
void iniciarEncoders(){
    bus[ENC_IZQ] = &Wire;// el encoder izquierdo habla por el bus 0 (& singnifica apunta en este bus)
    bus[ENC_DER] = &Wire1;// bus 1 

    for (uint8_t i = 0; i < NUM_ENC; i++) { //for para incio
        uint16_t v;
        activo[i] = leerRaw(*bus[i], v);        //responde o no
        angulo[i] = activo[i] ? v : 0;
        Serial.printf("ENC %d: %s\n", i, activo[i] ? "OK" : "ERROR"); //imprimir error o OK
    }
}

void actualizarEncoders(){
    for (uint8_t i = 0; i < NUM_ENC; i++) {
    if (!activo[i]) continue;// si no respondio al inicio, lo salta
    uint16_t v;//arbitrario v
    if (leerRaw(*bus[i], v)) angulo[i] = v;// si falla, conserva el ultimo valor
    }
}

uint16_t anguloRawEncoder(uint8_t indice){
    return angulo[indice];
} 

bool encoderActivo(uint8_t indice){
    return angulo[indice];
}

//vlx.cpp
#include "vlx.h"
#include "../config/config.h" // .. Salir de carpeta
#include <Wire.h> //I2C
#include <Adafruit_VL53L0X.h> //libraria VLX53L0X


static Adafruit_VL53L0X sensores[NUM_VLX];
static bool activo[NUM_VLX];
static uint16_t dist[NUM_VLX];

void iniciarVLX(){
    for (uint8_t i = 0; i < NUM_VLX; i++) { //ciclo for, cantidad de VLX sensores
    pinMode(VLX_XSHUT[i], OUTPUT); //set como output salida
    digitalWrite(VLX_XSHUT[i], LOW); //escribir, low
    activo[i] = false; //activo o no Sirve o no
    dist[i] = 0xFFFF; //valor inicial
  }
  
  delay(10);

  for (uint8_t i = 0; i < NUM_VLX; i++) {
    bool ok = false;
    for (uint8_t intento = 0; intento < 3 && !ok; intento++) {
      digitalWrite(VLX_XSHUT[i], HIGH);
      delay(30);
      ok = sensores[i].begin(VLX_DIR[i], false, &Wire);
      if (!ok) { digitalWrite(VLX_XSHUT[i], LOW); delay(10); }
    }
    if (ok) {
      activo[i] = true;
      sensores[i].startRangeContinuous(25);
    }
    Serial.printf("VLX %d: %s\n", i, ok ? "OK" : "ERROR"); //print si hay error
  }
}

void actualizarVLX(){
    for (uint8_t i = 0; i < NUM_VLX; i++) {
    if (!activo[i]) continue;
    if (sensores[i].isRangeComplete()) {
      uint16_t d = sensores[i].readRange();
      dist[i] = (d >= 8000) ? 0xFFFF : d;
      /*  Lo mismo, si se cumple lo que dice hace lo primero si no lo segundo
      if (d >= 8000) {
      dist[i] = 0xFFFF;
      } else {
      dist[i] = d;
      }
      */
    }
  }
}

uint16_t distanciaVLX(uint8_t i) { return dist[i]; } //regresa valor
